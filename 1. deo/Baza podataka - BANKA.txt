/* ==========================================================================
   DROP NAREDBE (bezbedne za ponovno pokretanje skripte - ne baca gresku
   ako objekat ne postoji)
   Redosled brisanja tabela je OBRNUT od redosleda kreiranja (prvo "dete",
   pa tek onda "roditelj") zbog FOREIGN KEY zavisnosti.
   ========================================================================== */

BEGIN
    FOR t IN (
        SELECT table_name FROM user_tables
        WHERE table_name IN (
            'SIGURNOSNA_KONTROLA','KAMATA','TRANSAKCIJA','KREDIT','DEPOZIT',
            'DEVIZNI_VALUTA','DEVIZNI_OGRANICENJE','STEDNI_BONUS',
            'STEDNI_USLOVI_PODIZANJA','TEKUCI_PAKET','ZIRO','DEVIZNI',
            'STEDNI','TEKUCI','RACUN','PRAVNO_LICE','FIZICKO_LICE'
        )
    ) LOOP
        EXECUTE IMMEDIATE 'DROP TABLE ' || t.table_name || ' CASCADE CONSTRAINTS PURGE';
    END LOOP;
END;
/

BEGIN
    FOR s IN (
        SELECT sequence_name FROM user_sequences
        WHERE sequence_name IN (
            'SEQ_FIZICKO_LICE','SEQ_PRAVNO_LICE','SEQ_RACUN','SEQ_TEKUCI_PAKET',
            'SEQ_STEDNI_USLOVI_PODIZANJA','SEQ_STEDNI_BONUS',
            'SEQ_DEVIZNI_OGRANICENJE','SEQ_DEVIZNI_VALUTA','SEQ_DEPOZIT',
            'SEQ_KREDIT','SEQ_TRANSAKCIJA','SEQ_KAMATA','SEQ_SIGURNOSNA_KONTROLA'
        )
    ) LOOP
        EXECUTE IMMEDIATE 'DROP SEQUENCE ' || s.sequence_name;
    END LOOP;
END;
/


/* ==========================================================================
   CREATE TABLE naredbe (Id je obican NUMBER, generisanje kljuca radi
   sekvenca + triger, umesto GENERATED BY DEFAULT AS IDENTITY)
   ========================================================================== */

CREATE TABLE FIZICKO_LICE (
    Id NUMBER PRIMARY KEY,
    Ime NVARCHAR2(100) NOT NULL,
    Prezime NVARCHAR2(100) NOT NULL,
    Jmbg VARCHAR2(13) UNIQUE,
    BrojLicneKarte VARCHAR2(20) UNIQUE,
    DatumRodjenja DATE,
    Adresa NVARCHAR2(255),
    Grad NVARCHAR2(100),
    Telefon VARCHAR2(20),
    Email VARCHAR2(100),
    Status NVARCHAR2(50),
    Komentar NVARCHAR2(1000)
);

CREATE TABLE PRAVNO_LICE (
    Id NUMBER PRIMARY KEY,
    NazivFirme NVARCHAR2(200) NOT NULL,
    Pib VARCHAR2(11) UNIQUE,
    Adresa NVARCHAR2(255),
    Grad NVARCHAR2(100),
    Telefon VARCHAR2(20),
    Email VARCHAR2(100),
    Status NVARCHAR2(50),
    Komentar NVARCHAR2(1000)
);

CREATE TABLE RACUN (
    Id NUMBER PRIMARY KEY,
    BrojRacuna VARCHAR2(20) UNIQUE NOT NULL,
    Valuta VARCHAR2(3) NOT NULL,
    TrenutnoStanje DECIMAL(18,2) DEFAULT 0,
    DatumOtvaranja DATE NOT NULL,
    Status NVARCHAR2(50),
    DozvoljeniMinus DECIMAL(18,2) DEFAULT 0,
    Komentar NVARCHAR2(1000),
    TipRacuna VARCHAR2(20) NOT NULL,
    KamatnaStopa DECIMAL(5,2),
    FizickoLiceId INT NULL,
    PravnoLiceId INT NULL,

    CONSTRAINT CHK_Racun_Klijent CHECK (
        (FizickoLiceId IS NOT NULL AND PravnoLiceId IS NULL) OR
        (FizickoLiceId IS NULL AND PravnoLiceId IS NOT NULL)
    ),
    FOREIGN KEY (FizickoLiceId) REFERENCES FIZICKO_LICE(Id),
    FOREIGN KEY (PravnoLiceId) REFERENCES PRAVNO_LICE(Id)
);

CREATE TABLE TEKUCI (
    RacunId INT PRIMARY KEY,
    PlatnaKartica NUMBER(1) DEFAULT 0 NOT NULL,
    MesecniLimit DECIMAL(18,2),
    CONSTRAINT CHK_PlatnaKartica CHECK (PlatnaKartica IN (0, 1)),
    FOREIGN KEY (RacunId) REFERENCES RACUN(Id)
);

CREATE TABLE STEDNI (
    RacunId INT PRIMARY KEY,
    MinimalniIznosOtvaranja DECIMAL(18,2),
    FrekvKapitalizKamate NUMBER(3) DEFAULT 12 NOT NULL,
    CONSTRAINT CHK_Stedni_FrekvKamate CHECK (FrekvKapitalizKamate IN (365, 12, 4, 2, 1)),
    FOREIGN KEY (RacunId) REFERENCES RACUN(Id)
);

CREATE TABLE DEVIZNI (
    RacunId INT PRIMARY KEY,
    Namena NVARCHAR2(100),
    KursnaRazlika DECIMAL(10,4),
    CONSTRAINT CHK_Namena CHECK (Namena IN ('ŠTEDNJA', 'POSLOVANJE')),
    FOREIGN KEY (RacunId) REFERENCES RACUN(Id)
);

CREATE TABLE ZIRO (
    RacunId INT PRIMARY KEY,
    Namena NVARCHAR2(100),
    ElektronskoBankarstvo NUMBER(1) DEFAULT 0 NOT NULL,
    LimitZaMasovnaPlacanja DECIMAL(18,2),
    IntegracijaSaSistemima NVARCHAR2(1000),
    CONSTRAINT CHK_ElektronskoBankarstvo CHECK (ElektronskoBankarstvo IN (0, 1)),
    FOREIGN KEY (RacunId) REFERENCES RACUN(Id)
);

CREATE TABLE TEKUCI_PAKET (
    Id NUMBER PRIMARY KEY,
    Paket NVARCHAR2(100) NOT NULL,
    RacunId INT NOT NULL,
    FOREIGN KEY (RacunId) REFERENCES RACUN(Id)
);

CREATE TABLE STEDNI_USLOVI_PODIZANJA (
    Id NUMBER PRIMARY KEY,
    UslovPodizanja NVARCHAR2(255) NOT NULL,
    RacunId INT NOT NULL,
    FOREIGN KEY (RacunId) REFERENCES RACUN(Id)
);

CREATE TABLE STEDNI_BONUS (
    Id NUMBER PRIMARY KEY,
    Bonus NVARCHAR2(255) NOT NULL,
    RacunId INT NOT NULL,
    FOREIGN KEY (RacunId) REFERENCES RACUN(Id)
);

CREATE TABLE DEVIZNI_OGRANICENJE (
    Id NUMBER PRIMARY KEY,
    Ogranicenje NVARCHAR2(255) NOT NULL,
    RacunId INT NOT NULL,
    FOREIGN KEY (RacunId) REFERENCES RACUN(Id)
);

CREATE TABLE DEVIZNI_VALUTA (
    Id NUMBER PRIMARY KEY,
    DozvoljenaValuta VARCHAR2(3) NOT NULL,
    RacunId INT NOT NULL,
    FOREIGN KEY (RacunId) REFERENCES RACUN(Id)
);

-- Napomena: OcekivanaKamata je IZVEDENI ATRIBUT
-- Formula: Iznos * (KamatnaStopa/100) * (PeriodOrocenja/12)
CREATE TABLE DEPOZIT (
    Id NUMBER PRIMARY KEY,
    DatumPocetka DATE NOT NULL,
    PeriodOrocenja INT NOT NULL,
    DatumIsteka DATE NOT NULL,
    StatusDepozita NVARCHAR2(50),
    Valuta VARCHAR2(3) NOT NULL,
    Iznos DECIMAL(18,2) NOT NULL,
    KamatnaStopa DECIMAL(5,2) NOT NULL,
    Komentar NVARCHAR2(1000),
    FizickoLiceId INT NULL,
    PravnoLiceId INT NULL,
    RacunId INT NOT NULL,

    CONSTRAINT CHK_Depozit_Klijent CHECK (FizickoLiceId IS NOT NULL OR PravnoLiceId IS NOT NULL),
    FOREIGN KEY (FizickoLiceId) REFERENCES FIZICKO_LICE(Id),
    FOREIGN KEY (PravnoLiceId) REFERENCES PRAVNO_LICE(Id),
    FOREIGN KEY (RacunId) REFERENCES RACUN(Id)
);

CREATE TABLE KREDIT (
    Id NUMBER PRIMARY KEY,
    DatumDospeca DATE NOT NULL,
    DatumOdobrenja DATE NOT NULL,
    Iznos DECIMAL(18,2) NOT NULL,
    Valuta VARCHAR2(3) NOT NULL,
    StatusKredita NVARCHAR2(50),
    MesecnaRata DECIMAL(18,2) NOT NULL,
    RokOtplate INT NOT NULL,             -- u mesecima
    Namena NVARCHAR2(255),
    KamatnaStopa DECIMAL(5,2) NOT NULL,
    Komentar NVARCHAR2(1000),
    FizickoLiceId INT NULL,
    PravnoLiceId INT NULL,
    RacunId INT NOT NULL,

    CONSTRAINT CHK_Kredit_Klijent CHECK (FizickoLiceId IS NOT NULL OR PravnoLiceId IS NOT NULL),
    FOREIGN KEY (FizickoLiceId) REFERENCES FIZICKO_LICE(Id),
    FOREIGN KEY (PravnoLiceId) REFERENCES PRAVNO_LICE(Id),
    FOREIGN KEY (RacunId) REFERENCES RACUN(Id)
);

CREATE TABLE TRANSAKCIJA (
    Id NUMBER PRIMARY KEY,
    DatumIVreme TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    TipTransakcije NVARCHAR2(50) NOT NULL,
    StatusTransakcije NVARCHAR2(50),
    PodaciPrimaoca NVARCHAR2(255),
    Referenca VARCHAR2(50),
    Valuta VARCHAR2(3) NOT NULL,
    Iznos DECIMAL(18,2) NOT NULL,
    Opis NVARCHAR2(255),
    Komentar NVARCHAR2(1000),
    RacunId INT NOT NULL,

    CONSTRAINT CHK_TipTransakcije CHECK (TipTransakcije IN ('UPLATA', 'ISPLATA', 'TRANSFER', 'PLAĆANJE_RAČUNA', 'KONVERZIJA')),
    FOREIGN KEY (RacunId) REFERENCES RACUN(Id)
);

CREATE TABLE KAMATA (
    Id NUMBER PRIMARY KEY,
    DatumObracuna DATE NOT NULL,
    PeriodObracuna NVARCHAR2(50),
    TipKamate NVARCHAR2(50) NOT NULL,
    StatusKamate NVARCHAR2(50),
    Iznos DECIMAL(18,2) NOT NULL,
    KreditId INT NULL,
    DepozitId INT NULL,
    RacunId INT NULL,

    CONSTRAINT CHK_StatusKamate CHECK (StatusKamate IN ('OBRAČUNATO','ISPLAĆENO','KAPITALIZOVANO')),
    -- Mora da se odnosi na barem jedan od tri izvora
    CONSTRAINT CHK_Kamata_Izvor CHECK (KreditId IS NOT NULL OR DepozitId IS NOT NULL OR RacunId IS NOT NULL),
    FOREIGN KEY (KreditId) REFERENCES KREDIT(Id),
    FOREIGN KEY (DepozitId) REFERENCES DEPOZIT(Id),
    FOREIGN KEY (RacunId) REFERENCES RACUN(Id)
);

CREATE TABLE SIGURNOSNA_KONTROLA (
    Id NUMBER PRIMARY KEY,
    IpAdresa VARCHAR2(45) NOT NULL,
    DatumIVreme TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    TipDogadjaja NVARCHAR2(50) NOT NULL,
    StatusDogadjaja NVARCHAR2(50),
    PodaciUredjaja NVARCHAR2(255),
    Opis NVARCHAR2(1000),
    RacunId INT NOT NULL,

    CONSTRAINT CHK_TipDogadjaja CHECK (TipDogadjaja IN ('LOGIN', 'NEUSPELA_AUTENTIFIKACIJA', 'PROMENA_PINA', 'BLOKADA_RAČUNA', 'SUMNJIVA_TRANSAKCIJA')),
    FOREIGN KEY (RacunId) REFERENCES RACUN(Id)
);

/* ==========================================================================
   TRIGERI
   Svaki triger puni Id iz odgovarajuce sekvence SAMO ako Id nije
   eksplicitno prosledjen u INSERT-u (:NEW.Id IS NULL) - zato mogu i
   dole navedeni INSERT-i sa rucno zadatim ID-jevima da rade normalno.
   ========================================================================== */

CREATE OR REPLACE TRIGGER TRG_FIZICKO_LICE
BEFORE INSERT ON FIZICKO_LICE
FOR EACH ROW
BEGIN
    IF :NEW.Id IS NULL THEN
        SELECT SEQ_FIZICKO_LICE.NEXTVAL INTO :NEW.Id FROM DUAL;
    END IF;
END;
/
